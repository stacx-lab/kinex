fact: {kind: "profile", device? (omit for the workspace manifest), pointer (JSON pointer), value? (exact match)} or {kind: "file", path (relative to .kinex)}

outcome: {status: done|partial|failed|needs-input, note, evidence: [path relative to .kinex]}
